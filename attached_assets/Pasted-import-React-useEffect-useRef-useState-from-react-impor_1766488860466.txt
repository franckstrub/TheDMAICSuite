import React, { useEffect, useRef, useState } from 'react';
import * as THREE from 'three';

const ResponseSurface3D = () => {
  const mountRef = useRef(null);
  const [view, setView] = useState('AB');
  const [showPoints, setShowPoints] = useState(true);
  const sceneRef = useRef(null);
  const cameraRef = useRef(null);
  const rendererRef = useRef(null);
  const animationRef = useRef(null);
  
  useEffect(() => {
    if (!mountRef.current) return;
    
    // Scene setup
    const scene = new THREE.Scene();
    scene.background = new THREE.Color(0xf0f0f0);
    sceneRef.current = scene;
    
    const camera = new THREE.PerspectiveCamera(
      50,
      mountRef.current.clientWidth / mountRef.current.clientHeight,
      0.1,
      1000
    );
    camera.position.set(4, 4, 4);
    camera.lookAt(0, 0, 0);
    cameraRef.current = camera;
    
    const renderer = new THREE.WebGLRenderer({ antialias: true });
    renderer.setSize(mountRef.current.clientWidth, mountRef.current.clientHeight);
    mountRef.current.appendChild(renderer.domElement);
    rendererRef.current = renderer;
    
    // Lights
    const ambientLight = new THREE.AmbientLight(0xffffff, 0.6);
    scene.add(ambientLight);
    
    const directionalLight = new THREE.DirectionalLight(0xffffff, 0.8);
    directionalLight.position.set(5, 10, 5);
    scene.add(directionalLight);
    
    // Quadratic model: y = 72 + 13.25*xA + 6.75*xB - 2.92*(xA² + xB² + xC²)
    const predict = (xA, xB, xC) => {
      return 72 + 13.25 * xA + 6.75 * xB - 2.92 * (xA*xA + xB*xB + xC*xC);
    };
    
    const createSurface = (viewType) => {
      const resolution = 40;
      const geometry = new THREE.BufferGeometry();
      const vertices = [];
      const colors = [];
      const indices = [];
      
      const scaleXY = 1.5;
      const scaleZ = 0.03;
      
      for (let i = 0; i <= resolution; i++) {
        for (let j = 0; j <= resolution; j++) {
          let x1, x2, x3;
          const t1 = -1 + (2 * i / resolution);
          const t2 = -1 + (2 * j / resolution);
          
          if (viewType === 'AB') {
            x1 = t1; x2 = t2; x3 = 0;
          } else if (viewType === 'AC') {
            x1 = t1; x2 = 0; x3 = t2;
          } else {
            x1 = 0; x2 = t1; x3 = t2;
          }
          
          const y = predict(x1, x2, x3);
          
          vertices.push(t1 * scaleXY, (y - 65) * scaleZ, t2 * scaleXY);
          
          const normalized = (y - 45) / (85 - 45);
          const color = new THREE.Color();
          color.setHSL(0.6 * normalized, 0.8, 0.5);
          colors.push(color.r, color.g, color.b);
        }
      }
      
      for (let i = 0; i < resolution; i++) {
        for (let j = 0; j < resolution; j++) {
          const a = i * (resolution + 1) + j;
          const b = a + resolution + 1;
          const c = a + 1;
          const d = b + 1;
          
          indices.push(a, b, c);
          indices.push(b, d, c);
        }
      }
      
      geometry.setAttribute('position', new THREE.Float32BufferAttribute(vertices, 3));
      geometry.setAttribute('color', new THREE.Float32BufferAttribute(colors, 3));
      geometry.setIndex(indices);
      geometry.computeVertexNormals();
      
      const material = new THREE.MeshPhongMaterial({
        vertexColors: true,
        side: THREE.DoubleSide,
        shininess: 30,
        transparent: true,
        opacity: 0.9
      });
      
      return new THREE.Mesh(geometry, material);
    };
    
    const createPoints = (viewType) => {
      const group = new THREE.Group();
      const scaleXY = 1.5;
      const scaleZ = 0.03;
      
      const factorialPoints = [
        { xA: -1, xB: -1, xC: 1, y: 45, color: 0xff4444 },
        { xA: 1, xB: -1, xC: -1, y: 68, color: 0xff4444 },
        { xA: -1, xB: 1, xC: -1, y: 55, color: 0xff4444 },
        { xA: 1, xB: 1, xC: 1, y: 85, color: 0xff4444 }
      ];
      
      const centralPoints = [
        { xA: 0, xB: 0, xC: 0, y: 71, color: 0x44ff44 },
        { xA: 0, xB: 0, xC: 0, y: 73, color: 0x44ff44 },
        { xA: 0, xB: 0, xC: 0, y: 72, color: 0x44ff44 }
      ];
      
      const allPoints = [...factorialPoints, ...centralPoints];
      
      allPoints.forEach(point => {
        let x, z;
        if (viewType === 'AB') {
          x = point.xA * scaleXY;
          z = point.xB * scaleXY;
        } else if (viewType === 'AC') {
          x = point.xA * scaleXY;
          z = point.xC * scaleXY;
        } else {
          x = point.xB * scaleXY;
          z = point.xC * scaleXY;
        }
        
        const sphereGeom = new THREE.SphereGeometry(0.08, 16, 16);
        const sphereMat = new THREE.MeshPhongMaterial({ color: point.color });
        const sphere = new THREE.Mesh(sphereGeom, sphereMat);
        sphere.position.set(x, (point.y - 65) * scaleZ, z);
        group.add(sphere);
        
        const lineGeom = new THREE.BufferGeometry().setFromPoints([
          new THREE.Vector3(x, -0.6, z),
          new THREE.Vector3(x, (point.y - 65) * scaleZ, z)
        ]);
        const lineMat = new THREE.LineBasicMaterial({ color: 0x999999 });
        const line = new THREE.Line(lineGeom, lineMat);
        group.add(line);
      });
      
      return group;
    };
    
    const createAxes = (viewType) => {
      const group = new THREE.Group();
      
      const axisLength = 2.0;
      const axesMaterial = new THREE.LineBasicMaterial({ color: 0x000000, linewidth: 2 });
      
      const xGeom = new THREE.BufferGeometry().setFromPoints([
        new THREE.Vector3(-axisLength, -0.6, 0),
        new THREE.Vector3(axisLength, -0.6, 0)
      ]);
      group.add(new THREE.Line(xGeom, axesMaterial));
      
      const yGeom = new THREE.BufferGeometry().setFromPoints([
        new THREE.Vector3(0, -0.6, 0),
        new THREE.Vector3(0, 1.0, 0)
      ]);
      group.add(new THREE.Line(yGeom, axesMaterial));
      
      const zGeom = new THREE.BufferGeometry().setFromPoints([
        new THREE.Vector3(0, -0.6, -axisLength),
        new THREE.Vector3(0, -0.6, axisLength)
      ]);
      group.add(new THREE.Line(zGeom, axesMaterial));
      
      const canvas = document.createElement('canvas');
      const context = canvas.getContext('2d');
      canvas.width = 256;
      canvas.height = 128;
      
      const createLabel = (text, position) => {
        context.clearRect(0, 0, canvas.width, canvas.height);
        context.fillStyle = '#000000';
        context.font = 'Bold 48px Arial';
        context.textAlign = 'center';
        context.fillText(text, 128, 80);
        
        const texture = new THREE.CanvasTexture(canvas);
        const spriteMaterial = new THREE.SpriteMaterial({ map: texture });
        const sprite = new THREE.Sprite(spriteMaterial);
        sprite.position.copy(position);
        sprite.scale.set(0.5, 0.25, 1);
        return sprite;
      };
      
      let labelX, labelZ;
      if (viewType === 'AB') {
        labelX = 'A (Temp)';
        labelZ = 'B (Conc)';
      } else if (viewType === 'AC') {
        labelX = 'A (Temp)';
        labelZ = 'C (Time)';
      } else {
        labelX = 'B (Conc)';
        labelZ = 'C (Time)';
      }
      
      group.add(createLabel(labelX, new THREE.Vector3(2.2, -0.6, 0)));
      group.add(createLabel('Yield', new THREE.Vector3(0, 1.2, 0)));
      group.add(createLabel(labelZ, new THREE.Vector3(0, -0.6, 2.2)));
      
      return group;
    };
    
    const gridHelper = new THREE.GridHelper(3, 10, 0xcccccc, 0xeeeeee);
    gridHelper.position.y = -0.6;
    scene.add(gridHelper);
    
    let surface = createSurface(view);
    let points = createPoints(view);
    let axes = createAxes(view);
    
    scene.add(surface);
    if (showPoints) scene.add(points);
    scene.add(axes);
    
    let angle = 0;
    const animate = () => {
      animationRef.current = requestAnimationFrame(animate);
      angle += 0.005;
      camera.position.x = 4 * Math.cos(angle);
      camera.position.z = 4 * Math.sin(angle);
      camera.lookAt(0, 0, 0);
      renderer.render(scene, camera);
    };
    animate();
    
    const handleResize = () => {
      if (mountRef.current && camera && renderer) {
        camera.aspect = mountRef.current.clientWidth / mountRef.current.clientHeight;
        camera.updateProjectionMatrix();
        renderer.setSize(mountRef.current.clientWidth, mountRef.current.clientHeight);
      }
    };
    window.addEventListener('resize', handleResize);
    
    return () => {
      window.removeEventListener('resize', handleResize);
      if (animationRef.current) {
        cancelAnimationFrame(animationRef.current);
      }
      if (mountRef.current && renderer.domElement) {
        mountRef.current.removeChild(renderer.domElement);
      }
      surface.geometry.dispose();
      surface.material.dispose();
      points.traverse(child => {
        if (child.geometry) child.geometry.dispose();
        if (child.material) child.material.dispose();
      });
    };
  }, [view, showPoints]);
  
  return (
    <div className="w-full h-screen bg-gray-50 p-4">
      <div className="max-w-7xl mx-auto h-full flex flex-col">
        <div className="bg-white rounded-lg shadow-lg p-4 mb-4">
          <h1 className="text-2xl font-bold text-gray-800 mb-2">
            3D Response Surface - 2^(3-1) Factorial Design
          </h1>
          <p className="text-gray-600 mb-4">
            Model: Yield = 72 + 13.25·A + 6.75·B - 2.92·(A² + B² + C²)
          </p>
          
          <div className="flex flex-wrap gap-4 items-center">
            <div className="flex gap-2">
              <span className="font-semibold text-gray-700">View:</span>
              <button
                onClick={() => setView('AB')}
                className={`px-4 py-2 rounded ${
                  view === 'AB' 
                    ? 'bg-blue-600 text-white' 
                    : 'bg-gray-200 text-gray-700 hover:bg-gray-300'
                }`}
              >
                A vs B (Temp vs Conc)
              </button>
              <button
                onClick={() => setView('AC')}
                className={`px-4 py-2 rounded ${
                  view === 'AC' 
                    ? 'bg-blue-600 text-white' 
                    : 'bg-gray-200 text-gray-700 hover:bg-gray-300'
                }`}
              >
                A vs C (Temp vs Time)
              </button>
              <button
                onClick={() => setView('BC')}
                className={`px-4 py-2 rounded ${
                  view === 'BC' 
                    ? 'bg-blue-600 text-white' 
                    : 'bg-gray-200 text-gray-700 hover:bg-gray-300'
                }`}
              >
                B vs C (Conc vs Time)
              </button>
            </div>
            
            <div className="flex items-center gap-2">
              <input
                type="checkbox"
                id="showPoints"
                checked={showPoints}
                onChange={(e) => setShowPoints(e.target.checked)}
                className="w-4 h-4"
              />
              <label htmlFor="showPoints" className="text-gray-700">
                Show experimental points
              </label>
            </div>
          </div>
          
          <div className="mt-4 flex gap-6 text-sm">
            <div className="flex items-center gap-2">
              <div className="w-4 h-4 bg-red-500 rounded-full"></div>
              <span>Factorial points (4)</span>
            </div>
            <div className="flex items-center gap-2">
              <div className="w-4 h-4 bg-green-500 rounded-full"></div>
              <span>Center points (3)</span>
            </div>
          </div>
        </div>
        
        <div 
          ref={mountRef} 
          className="flex-1 bg-white rounded-lg shadow-lg overflow-hidden"
          style={{ minHeight: '600px' }}
        />
        
        <div className="bg-white rounded-lg shadow-lg p-4 mt-4">
          <h3 className="font-bold text-gray-800 mb-2">Interpretation:</h3>
          <ul className="text-sm text-gray-700 space-y-1">
            <li>• <strong>Color:</strong> Blue = low yield, Green/Yellow = high yield</li>
            <li>• <strong>Visible curvature:</strong> Surface is curved (significant quadratic effect)</li>
            <li>• <strong>Optimum:</strong> Central zone near (0, 0) → 100°C, 4%, 45min → Yield ≈ 72%</li>
            <li>• <strong>Gradient:</strong> Effect of A (Temperature) is stronger than B (Concentration)</li>
          </ul>
        </div>
      </div>
    </div>
  );
};

export default ResponseSurface3D;
